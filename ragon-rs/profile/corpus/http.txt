//! A minimal HTTP/1.1 client over loopback TCP: GET, and a streamed POST that reports each
//! server-sent `data:` line with its arrival time. No TLS: the model server answers on loopback.
use std::io::{BufRead, BufReader, Read, Write};
use std::net::TcpStream;
use std::time::{Duration, Instant};

/// "http://127.0.0.1:8000/" or "127.0.0.1:8000" to "127.0.0.1:8000".
pub fn host_port(url: &str) -> String {
    let u = url.trim().trim_start_matches("http://");
    u.split('/').next().unwrap_or(u).to_string()
}

fn connect(addr: &str, timeout: Duration) -> Result<TcpStream, String> {
    let s = TcpStream::connect(addr).map_err(|e| format!("cannot connect to {addr}: {e}"))?;
    s.set_read_timeout(Some(timeout)).ok();
    s.set_write_timeout(Some(timeout)).ok();
    s.set_nodelay(true).ok();
    Ok(s)
}

fn send(stream: &mut TcpStream, method: &str, addr: &str, path: &str, body: Option<&str>) -> Result<(), String> {
    let mut req = format!("{method} {path} HTTP/1.1\r\nHost: {addr}\r\nConnection: close\r\nAccept: */*\r\n");
    if let Some(b) = body {
        req += &format!("Content-Type: application/json\r\nContent-Length: {}\r\n", b.len());
    }
    req += "\r\n";
    if let Some(b) = body {
        req += b;
    }
    stream.write_all(req.as_bytes()).map_err(|e| format!("write failed: {e}"))
}

/// Reads the status line and headers; returns (status, chunked, content_length).
fn read_head(r: &mut BufReader<TcpStream>) -> Result<(u16, bool, Option<usize>), String> {
    let mut line = String::new();
    r.read_line(&mut line).map_err(|e| format!("read failed: {e}"))?;
    let status = line.split_whitespace().nth(1).and_then(|s| s.parse().ok()).ok_or("bad status line")?;
    let (mut chunked, mut len) = (false, None);
    loop {
        line.clear();
        if r.read_line(&mut line).map_err(|e| format!("read failed: {e}"))? == 0 || line.trim().is_empty() {
            break;
        }
        let l = line.to_ascii_lowercase();
        if let Some(v) = l.strip_prefix("transfer-encoding:") {
            chunked = v.contains("chunked");
        } else if let Some(v) = l.strip_prefix("content-length:") {
            len = v.trim().parse().ok();
        }
    }
    Ok((status, chunked, len))
}

/// Feeds the body to `on_bytes` as it arrives, decoding chunked transfer.
fn read_body(r: &mut BufReader<TcpStream>, chunked: bool, len: Option<usize>, on_bytes: &mut dyn FnMut(&[u8])) -> Result<(), String> {
    if chunked {
        loop {
            let mut size = String::new();
            if r.read_line(&mut size).map_err(|e| format!("read failed: {e}"))? == 0 {
                return Ok(());
            }
            let n = usize::from_str_radix(size.trim().split(';').next().unwrap_or("0"), 16).map_err(|_| "bad chunk size")?;
            if n == 0 {
                return Ok(());
            }
            let mut buf = vec![0u8; n];
            r.read_exact(&mut buf).map_err(|e| format!("read failed: {e}"))?;
            on_bytes(&buf);
            let mut crlf = [0u8; 2];
            r.read_exact(&mut crlf).ok();
        }
    }
    let mut left = len.unwrap_or(usize::MAX);
    let mut buf = [0u8; 8192];
    while left > 0 {
        let want = buf.len().min(left);
        let n = r.read(&mut buf[..want]).map_err(|e| format!("read failed: {e}"))?;
        if n == 0 {
            break;
        }
        on_bytes(&buf[..n]);
        left -= n;
    }
    Ok(())
}

pub fn get(addr: &str, path: &str, timeout: Duration) -> Result<String, String> {
    let mut s = connect(addr, timeout)?;
    send(&mut s, "GET", addr, path, None)?;
    let mut r = BufReader::new(s);
    let (status, chunked, len) = read_head(&mut r)?;
    let mut body = Vec::new();
    read_body(&mut r, chunked, len, &mut |b| body.extend_from_slice(b))?;
    if status != 200 {
        return Err(format!("GET {path}: HTTP {status}"));
    }
    Ok(String::from_utf8_lossy(&body).into_owned())
}

/// POSTs `body` and calls `on_data(payload, arrival)` for every `data: …` line of the event stream
/// (including `[DONE]`). Arrival is the instant the bytes carrying the line were read.
pub fn post_stream(addr: &str, path: &str, body: &str, timeout: Duration, on_data: &mut dyn FnMut(&str, Instant)) -> Result<(), String> {
    let mut s = connect(addr, timeout)?;
    send(&mut s, "POST", addr, path, Some(body))?;
    let mut r = BufReader::new(s);
    let (status, chunked, len) = read_head(&mut r)?;
    if status != 200 {
        let mut b = Vec::new();
        read_body(&mut r, chunked, len, &mut |x| b.extend_from_slice(x)).ok();
        return Err(format!("POST {path}: HTTP {status} {}", String::from_utf8_lossy(&b).chars().take(200).collect::<String>()));
    }
    let mut pending = String::new();
    read_body(&mut r, chunked, len, &mut |bytes| {
        let at = Instant::now();
        pending.push_str(&String::from_utf8_lossy(bytes));
        while let Some(i) = pending.find('\n') {
            let line: String = pending.drain(..=i).collect();
            if let Some(p) = line.trim_end().strip_prefix("data:") {
                on_data(p.trim_start(), at);
            }
        }
    })
}
